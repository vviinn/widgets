# One-click Discovery

One-click discovery allows you to display products with a single click inside the modal or in appended mode. Here are examples using the `vviinn-results` component or separately.

<div id="one-click-modal">
    <div className="tab tab-reverse">
        <h3>One-click Discovery Modal</h3>
        <div className="tablink-group">
            <button className="tablink tablink--sm" onClick={(e)=> {clickHandlers(e, 'Code_1', 'one-click-modal')}}>Code</button>
            <button className="tablink tablink--sm active" onClick={(e)=> {clickHandlers(e, 'Preview_1', 'one-click-modal')}}>Preview</button>
        </div>
    </div>
    <div className="tab-sub-title">One-click Discovery's `modal` mode, showing in `vviinn-results` for the "{textSearchQuery}" query.</div>
    <div id="Preview_1" className="tabcontent" style={{display:"block"}}>
        <div className="tabcontent-preview">
            <vviinn-results 
                token={vviinnToken} 
                image-resolution-width="500" 
                currency-sign="€" 
                locale="en-US" 
                results-count="6"
                text-search-query={textSearchQuery}
                one-click-discovery-mode="modal"
            ></vviinn-results>
        </div>
    </div>
    <div id="Code_1" className="tabcontent">
```html
<vviinn-results 
    token="TOKEN" 
    image-resolution-width="500"

    locale="en-US"

    text-search-query="EXAMPLE_QUERY"
    one-click-discovery-mode="modal"
></vviinn-results>
```
    </div>
</div>
