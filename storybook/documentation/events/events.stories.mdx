## Listen to Events. Examples

### Recommendation Widget

```javascript
document.addEventListener("vviinnProductLoad", ({ detail }) => {
  console.log(detail);
});

// Output:
{
    "productId": "123456",
    "productRank": 0,
    "productName": "Couchtisch Rolf Benz 8770",
    "productType": "Table",
    "widgetType": "VPR",
    "campaignTypeId": "VPR",
    "campaignTypeName": "Similar Products",
    "widgetId": "vpr-widget-1",
    "widgetVersion": "2.1.0"
}
// ... and other loaded products
```

### Search Button

```javascript

document.addEventListener("vviinnSelectFilter", ({ detail }) => {
  console.log(detail);
});

// Output: 
{
    "widgetType": "VPS",
    "campaignTypeId": "VPS",
    "campaignTypeName": "Visual Search",
    "widgetId": "vps-button-1",
    "action": "select",
    "filterName": "Vintage Teppiche",
    "widgetVersion": "2.1.0"
}
```
