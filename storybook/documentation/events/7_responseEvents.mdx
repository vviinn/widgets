# Response Events

These events are triggered by backend responses and contain instructions or suggestions from the API server.

These props are available for each event:

- campaignTypeId
- campaignTypeName
- widgetId (if exists)
- widgetVersion

Some events have additional props, summarized in the table below.

## Listen to Events. Examples

### Basic Usage of Redirect Event

```javascript
document.addEventListener("vviinnRedirect", ({ detail }) => {
  console.log(detail);
  // Widget will automatically redirect unless intercepted
});

// Output (auto-suggest):
{
    "url": "https://www.my-site.de/sansibar",
    "source": "auto-suggest",
    "query": "sansibar",
    "campaignTypeId": "TPS",
    "campaignTypeName": "Text Product Search",
    "widgetId": "my-search-widget",
    "widgetVersion": "2.205.0"
}

// Output (text search with results-page-url):
{
    "url": "https://www.my-site.de/results?query=shoes",
    "source": "text-search",
    "query": "shoes",
    "campaignTypeId": "TPS",
    "campaignTypeName": "Text Product Search",
    "widgetVersion": "2.205.0"
}

// Output (image search with results-page-url):
{
    "url": "https://www.my-site.de/results?searchId=abc-123",
    "source": "image-search",
    "searchId": "abc-123",
    "campaignTypeId": "VPS",
    "campaignTypeName": "Visual Search",
    "widgetVersion": "2.205.0"
}
```

### Intercept Redirect

You can intercept the redirect by calling `event.preventDefault()` to handle navigation yourself.

```javascript
document.addEventListener("vviinnRedirect", (event) => {
  event.preventDefault();

  const { url, query } = event.detail;

  // Your custom logic: SPA routing, analytics, confirmation dialogs, etc.
  myRouter.push(url);
});
```

### Basic Usage of Search Success Event

```javascript
document.addEventListener("vviinnSearchSuccess", ({ detail }) => {
  console.log(detail);
  // Use results data for custom UI implementation
});

// Output example (Visual Search):
{
    "results": [
        {
            "productId": "12345",
            "title": "Product Name",
            "price": { "actual": { "amount": 29.99 } },
            "image": { "url": "https://..." }
            // ... more product fields
        }
        // ... more products
    ],
    "dynamicFilters": [
        {
            "kind": "color",
            "values": [
                { "value": "red", "count": 15 },
                { "value": "blue", "count": 8 }
            ]
        }
    ],
    "detectedObjects": [
        { "left": 0.1, "top": 0.2, "width": 0.3, "height": 0.4 }
    ],
    "requestId": "abc123-def456",
    "storeName": "widget",
    "campaignTypeId": "VPS",
    "campaignTypeName": "Visual Search",
    "widgetVersion": "2.208.0"
}
```

### Basic Usage of No Result Event

```javascript
document.addEventListener("vviinnNoResult", ({ detail }) => {
  console.log(detail);
  // Handle error - show notification, retry, etc.
});

// Output:
{
    "campaignTypeId": "TPS",
    "campaignTypeName": "Text Product Search",
    "widgetId": "my-search-widget",
    "widgetVersion": "2.208.0"
}
```
